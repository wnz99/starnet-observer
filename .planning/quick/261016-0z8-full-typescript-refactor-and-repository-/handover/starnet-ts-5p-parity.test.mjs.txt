function dom(document) {
  return { createElement: (tag) => document.createElement(tag), children: (node) => node.children,
    attach: (parent, ...nodes) => parent.append(...nodes), replaceChildren: (parent, ...nodes) => parent.replaceChildren(...nodes),
    insertBefore: (parent, node, before) => parent.insertBefore(node, before) };
}
