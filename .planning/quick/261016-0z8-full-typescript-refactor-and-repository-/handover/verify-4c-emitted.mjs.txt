import fs from "node:fs/promises";
import path from "node:path";
import os from "node:os";
import { spawnSync } from "node:child_process";
import assert from "node:assert/strict";
const repository="/Users/davidfava/Projects/Personal/starnet-observer-typescript-4c";
const runtime=path.join(repository,"dist/runtime");
const files=["src/sidecar/inspector/index.test.ts","src/sidecar/transcripts/index.test.ts","src/sidecar/helpers/claims.test.ts","src/sidecar/helpers/index.test.ts"];
const sandbox=await fs.mkdtemp(path.join(os.tmpdir(),"starnet-4c-emitted-"));
try {
  const tests=[];
  for(const [index,file] of files.entries()) {
    let source=await fs.readFile(path.join(repository,file),"utf8");
    for(const match of [...source.matchAll(/from "([.][^"]+)"/g)]) {
      const specifier=match[1],resolved=path.resolve(path.dirname(path.join(repository,file)),specifier);
      const target=specifier.includes("testing.fixture") ? resolved.replace(/\.js$/,".ts") : path.join(runtime,path.relative(repository,resolved));
      await fs.access(target);
      source=source.replaceAll('from "'+specifier+'"','from '+JSON.stringify(target));
    }
    const target=path.join(sandbox,"emitted-"+index+".test.ts");
    await fs.writeFile(target,source);
    tests.push(target);
  }
  const visited=new Set();
  async function audit(file) {
    if(visited.has(file)) return;
    visited.add(file);
    assert(file.startsWith(runtime+path.sep),"emitted application import escaped runtime");
    assert(file.endsWith(".js"),"application dependency must be emitted JavaScript");
    const source=await fs.readFile(file,"utf8");
    for(const match of source.matchAll(/(?:from\s*|import\s*)["'](\.[^"']+)["']/g)) await audit(path.resolve(path.dirname(file),match[1]));
  }
  for(const file of ["src/sidecar/transcripts/index.js","src/sidecar/inspector/index.js","src/sidecar/helpers/index.js"]) await audit(path.join(runtime,file));
  console.log("Emitted application closure:",visited.size,"JS modules; zero legacy/source TS application imports");
  const result=spawnSync(process.execPath,["--no-env-file","test",...tests],{cwd:sandbox,stdio:"inherit",env:process.env});
  assert.equal(result.status,0);
} finally {await fs.rm(sandbox,{recursive:true,force:true});}
