import assert from 'node:assert/strict';
import fs from 'node:fs/promises';
import os from 'node:os';
import path from 'node:path';
import {promisify} from 'node:util';
import {execFile} from 'node:child_process';
import {pathToFileURL} from 'node:url';
const root = process.argv[2];
const emitted = path.join(root, 'dist/runtime/src/sidecar');
const arrangement = await import(pathToFileURL(path.join(emitted, 'arrangement/index.js')));
const projects = await import(pathToFileURL(path.join(emitted, 'projects/index.js')));
const identity = await import(pathToFileURL(path.join(emitted, 'identity/index.js')));
const temp = await fs.mkdtemp(path.join(os.tmpdir(), 'starnet-leaves-emitted-'));
const executeFile = promisify(execFile);
try {
 const execute = (command, args, settings) => executeFile(command,args,{...settings,env:{PATH:process.env.PATH,HOME:temp,GIT_CONFIG_NOSYSTEM:'1',GIT_CONFIG_GLOBAL:'/dev/null'}});
 const first = path.join(temp,'workspace','first'), second = path.join(temp,'workspace','second');
 for (const [folder,name] of [[first,'first'],[second,'second']]) {
  await fs.mkdir(folder,{recursive:true});
  await execute('git',['-C',folder,'init','-q'],{});
  await execute('git',['-C',folder,'remote','add','origin',`https://github.com/fixture/${name}.git`],{});
 }
 const resolver = projects.makeProjectResolver({execute,now:()=>1000,home:temp,arrange:{home:temp}});
 const rows = await resolver.resolveMany([second,first,second,null,undefined]);
 assert.equal(rows.length,5);
 assert.equal(rows[0].arrangement.room,'fixture');
 assert.equal(rows[0].arrangement.roomId,rows[1].arrangement.roomId);
 assert.deepEqual(rows[0],rows[2]);
 assert.equal(rows[3].projectId,'unassigned');
 assert.equal(rows[4].projectId,'unassigned');
 assert(!JSON.stringify(rows).includes(temp));
 assert.deepEqual(arrangement.parseRemote('git@github.com:Fixture/First.git'),{host:'github.com',org:'fixture',name:'first'});
 assert.throws(()=>arrangement.validateRooms({rooms:[{name:'invalid',match:['../outside']}]}));
 const pane={pane_id:'fixture:p1',agent:'codex',terminal_id:'fixture_terminal'};
 const session='00000000-0000-0000-0000-000000000001';
 const methods=[];
 const identities=identity.makeIdentityResolver({now:()=>1000,readConfiguration:async()=> 'configured',readProcessBirth:async pid=>({pid,startSeconds:1,startMicroseconds:2}),rpc:async(method)=>{
  methods.push(method);
  if(method==='agent.get')return {result:{agent:pane}};
  if(method==='pane.process_info')return {result:{process_info:{pane_id:pane.pane_id,shell_pid:41}}};
  if(method==='pane.get')return {result:{pane:{pane_id:pane.pane_id,scroll:{max_offset_from_bottom:0,viewport_rows:20}}}};
  if(method==='pane.copy_motion')return {result:{pane_id:pane.pane_id,content_revision:8}};
  assert.equal(method,'pane.read');
  return {result:{read:{pane_id:pane.pane_id,source:'recent',format:'text',text:`\n  ${session} · model\n  ? for shortcuts\n`}}};
 }});
 const resolved=await identities.resolve(pane,{});
 assert.equal(resolved.source,'footer');
 assert.equal(resolved.sessionId,session);
 assert.equal(methods.at(-1),'pane.copy_motion');
 assert.equal(await identities.revalidate(resolved.proof),true);
 console.log('Emitted leaf smoke passed: real synthetic Git discovery/grouping/privacy, aligned invalid inputs, room validation, footer capture and revalidation.');
} finally {await fs.rm(temp,{recursive:true,force:true});}
